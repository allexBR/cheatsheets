#------------------------------------------------#
#  VMware Workstation Pro Suricata Configuration #
#------------------------------------------------#

NOTE: The following network settings are required for the Suricata VM (with Debian 13 installed) works in transparent bridge mode.


# Install bridge-utils package (required):
apt update && apt install -y bridge-utils


# Create the bridge device:
brctl addbr br0


# Check which interfaces exist on your computer:
ip addr show | grep ^[0-9]


# Add all the interfaces you want to the bridge:
brctl addif br0 eth0 eth1 ...


# Finally, bring your bridge up:
ifup br0



----------------------------------------------------------------------------------------

# Template Suricata VM (IDS inline - transparent bridge mode)
# NOTE: Set up interfaces manually, avoiding conflicts with, e.g., network manager.


nano /etc/network/interfaces


# The loopback network interface
auto lo
iface lo inet loopback

# The physical network interfaces

# Firewall side (VMnet5)
auto ens33
iface ens33 inet manual
     up ip link set $IFACE up
     up ip link set $IFACE promisc on

# Endpoints side (VMnet6)
auto ens34
iface ens34 inet manual
     up ip link set $IFACE up
     up ip link set $IFACE promisc on

# Transparent bridge mode (inline)
auto br0
iface br0 inet static
    address 192.168.1.100
    netmask 255.255.255.0
    gateway 192.168.1.1
    bridge_ports ens33 ens34   # bind to any ports
    bridge_stp off             # disable Spanning Tree Protocol
    bridge_waitport 0          # no delay before a port becomes available
    bridge_fd 0                # no forwarding delay
    bridge_maxwait 0           # if it is equal to 0 then no waiting is done
    up ip link set dev br0 up




# Validation (in order):

ip link show ens33
ip link show ens34

After:

ip link show br0
